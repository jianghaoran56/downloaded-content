cmake_minimum_required(VERSION 3.15)
project(ComputerProgram LANGUAGES CXX)
add_executable(ComputerProgram WIN32 src/main.cpp)
target_compile_features(ComputerProgram PRIVATE cxx_std_17)
target_compile_definitions(ComputerProgram PRIVATE UNICODE _UNICODE)
if(MSVC)
  target_compile_options(ComputerProgram PRIVATE /W3)
else()
  target_compile_options(ComputerProgram PRIVATE -Wall -Wextra)
endif()
